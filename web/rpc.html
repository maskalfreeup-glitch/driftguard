Developer Tooling

One-Click Integration Snippets

Drop-in configurations for Foundry, Hardhat, and MetaMask with zero desync risk.

Execute cast commands or run Forge deployment scripts without sequencer desync risk: 

# 1. Query verified canonical block height from Arbitrum One
cast block-number --rpc-url https://rpc.driftguard.live/arb

# 2. Query gas price with guaranteed monotonic consensus
cast gas-price --rpc-url https://rpc.driftguard.live/arb

# 3. Simulate or execute contract call with zero desync risk
cast call 0x912ce59144191c1204e64559fe8253a0e49e6548 "totalSupply()(uint256)" \
  --rpc-url https://rpc.driftguard.live/arb

# 4. Deploy contract script via DriftGuard sentry
forge script script/Deploy.s.sol \
  --rpc-url https://rpc.driftguard.live/arb \
  --broadcast --verify

Add DriftGuard network definitions to your hardhat.config.ts: 

import { HardhatUserConfig } from "hardhat/config";
import "@nomicfoundation/hardhat-toolbox";

const config: HardhatUserConfig = {
  solidity: "0.8.24",
  networks: {
    arbitrum: {
      url: "https://rpc.driftguard.live/arb",
      chainId: 42161,
      timeout: 20000,
    },
    arbitrumNova: {
      url: "https://rpc.driftguard.live/nova",
      chainId: 42170,
      timeout: 20000,
    },
    arbitrumSepolia: {
      url: "https://rpc.driftguard.live/arb-sepolia",
      chainId: 421614,
      timeout: 20000,
    },
    base: {
      url: "https://rpc.driftguard.live/base",
      chainId: 8453,
      timeout: 20000,
    },
  },
};

export default config;
